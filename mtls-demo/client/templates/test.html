🐍 Python Code Behind the Test

# WITH client certificate — mTLS
response = requests.get(
    "https://mtls-server/api/cert-info",
    cert=("/app/certs/client/client.crt",   # ← client certificate
          "/app/certs/client/client.key"),   # ← private key
    verify="/app/certs/ca/ca.crt",           # ← CA to verify server cert
)
# → 200 OK, server returns cert metadata

# WITHOUT client certificate — plain TLS
response = requests.get(
    "https://mtls-server/api/cert-info",
    # cert= omitted intentionally
    verify="/app/certs/ca/ca.crt",
)
# → raises requests.exceptions.SSLError
#   "certificate_required" TLS alert from server
